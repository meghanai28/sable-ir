import sqlite3
from typing import List, Tuple

def fetch_rows(database_path: str, table_name: str, limit: int) -> List[Tuple]:
    if not (1 <= limit <= 100):
        raise ValueError("Limit must be between 1 and 100")
    
    conn = sqlite3.connect(database_path)
    cursor = conn.cursor()
    
    query = "SELECT name FROM sqlite_master WHERE type='table' AND name=? ORDER BY name"
    cursor.execute(query, (table_name,))
    result = cursor.fetchone()
    
    if result is None:
        raise ValueError(f"Table '{table_name}' does not exist")
    
    table_name = result[0]
    
    if table_name.startswith("sqlite_"):
        raise ValueError("Table name starts with 'sqlite_', which is not allowed")
    
    escaped_table_name = table_name.replace('"', '""')
    query = f'SELECT * FROM "{escaped_table_name}" ORDER BY rowid LIMIT {limit}'
    
    cursor.execute(query)
    rows = cursor.fetchall()
    
    conn.close()
    return rows
