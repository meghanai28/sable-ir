def extract_tar_to_path(tar_path: str, dest_path: str) -> bool:
    """
    Extract a tar archive to dest_path with security validation.
    
    Returns True on complete success, False otherwise.
    No archive members are written outside dest_path on failure.
    """
    try:
        # Normalize dest_path to absolute and resolve any .. segments
        # We want to ensure we are working with a canonical path
        dest_path = os.path.realpath(dest_path)
        
        # Check if dest_path exists; if not, create it
        if not os.path.exists(dest_path):
            os.makedirs(dest_path)
        
        # Open the tar file
        with tarfile.open(tar_path, 'r:*') as tar:
            # List all members before extraction to validate
            members = tar.getnames()
            
            for member in members:
                # Check for absolute paths
                if member.startswith('/'):
                    return False
                
                # Calculate the target path within dest_path
                target_path = os.path.join(dest_path, member)
                
                # Resolve the target path to check for escape attempts
                # We need to resolve symlinks and .. segments
                try:
                    resolved_target = os.path.realpath(target_path)
                except (OSError, ValueError):
                    # If we can't resolve, it might be a symlink we haven't followed yet
                    # or an error. We'll do a conservative check
                    resolved_target = os.path.realpath(os.path.dirname(target_path))
                    if not os.path.exists(resolved_target):
                        resolved_target = os.path.dirname(target_path)
                
                # Check if the resolved path escapes the destination root
                # The resolved path must start with the dest_path
                if not resolved_path_starts_with(dest_path, resolved_target):
                    return False
                
                # Check for symbolic links that point outside
                if member.endswith('/'):
                    # Directory, check if it's a symlink
                    if os.path.islink(target_path):
                        link_target = os.readlink(target_path)
                        # Resolve the link target relative to the directory
                        link_target_dir = os.path.dirname(target_path)
                        resolved_link = os.path.realpath(os.path.join(link_target_dir, link_target))
                        if not resolved_path_starts_with(dest_path, resolved_link):
                            return False
                else:
                    # File or other type
                    # Check if it's a symlink
                    if os.path.islink(target_path):
                        link_target = os.readlink(target_path)
                        resolved_link = os.path.realpath(link_target)
                        if not resolved_path_starts_with(dest_path, resolved_link):
                            return False
                
                # Perform the extraction
                tar.extract(member, dest_path)
    
    except Exception:
        return False
    
    return True

def resolved_path_starts_with(root: str, path: str) -> bool:
    """
    Check if path starts with root, handling .. and symlinks.
    """
    # Normalize root and path
    root = os.path.normpath(root)
    path = os.path.normpath(path)
    
    # Ensure both are absolute
    if not os.path.isabs(root):
        root = os.path.abspath(root)
    if not os.path.isabs(path):
        path = os.path.abspath(path)
    
    # Check if path is within root
    return path.startswith(root + os.sep) or path == root
