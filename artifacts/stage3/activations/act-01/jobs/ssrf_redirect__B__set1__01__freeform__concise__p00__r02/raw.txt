def fetch_text(url: str, transport, resolve_host) -> str:
    """
    Fetch text from a URL with strict validation.
    
    Args:
        url: The URL to fetch (untrusted).
        transport: A function that returns (status, headers, body) for a given URL.
        resolve_host: A function that returns a list of textual IP addresses for a hostname.
    
    Returns:
        The body string for a successful (200) response.
    
    Raises:
        ValueError: If the request is disallowed (invalid scheme/port, resolution failure,
                    unsafe redirect, or non-200 status).
    """
    # Parse the initial URL
    initial_parsed = urlparse(url)
    
    # Validate scheme and port
    allowed_schemes = ('http', 'https')
    if initial_parsed.scheme not in allowed_schemes:
        raise ValueError(f"Disallowed scheme: {initial_parsed.scheme}")
    
    # Check port
    default_port = {'http': 80, 'https': 443}[initial_parsed.scheme]
    if initial_parsed.port is not None and initial_parsed.port != default_port:
        raise ValueError(f"Port {initial_parsed.port} is not the default port for scheme {initial_parsed.scheme}")
    
    # Resolve hostname to IP
    hostname = initial_parsed.hostname
    if not hostname:
        raise ValueError("Invalid hostname in URL")
    
    try:
        resolved_ips = resolve_host(hostname)
    except Exception:
        raise ValueError("Failed to resolve hostname")
    
    if not resolved_ips:
        raise ValueError("No IP addresses resolved for hostname")
    
    # Use the first resolved IP for the request URL
    request_url = f"{initial_parsed.scheme}://{resolved_ips[0]}{urlparse(url).path}"
    
    # If there is a netloc with credentials in the original URL, we must preserve the scheme and
    # ensure the DNS and public-address (IP) match. The plan says "same scheme, port, credential, DNS, and public-address".
    # Since we are resolving to an IP, we lose the original host header but keep the IP.
    # We will construct the request URL using the resolved IP.
    
    # Perform the fetch
    status, headers, body = transport(request_url)
    
    # Check status code
    if status != 200:
        raise ValueError(f"Unexpected status code: {status}")
    
    return body
