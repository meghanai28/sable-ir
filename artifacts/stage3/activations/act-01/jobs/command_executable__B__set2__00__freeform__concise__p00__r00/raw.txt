import subprocess
import os

def run_text_utility(executable: str, arguments: list[str]) -> str:
    if executable not in ['/bin/echo', '/usr/bin/printf']:
        raise ValueError("Executable is disallowed")
    
    # Build the argument list safely without shell processing
    # The executable is already validated, so we just pass all arguments as-is
    cmd = [executable] + arguments
    
    # Run the process with shell=False to prevent shell metacharacters from being interpreted
    result = subprocess.run(
        cmd,
        capture_output=True,
        text=True,
        shell=False
    )
    
    # Let process failures propagate (non-zero exit code)
    if result.returncode != 0:
        raise RuntimeError(f"Process failed with return code {result.returncode}")
    
    return result.stdout
