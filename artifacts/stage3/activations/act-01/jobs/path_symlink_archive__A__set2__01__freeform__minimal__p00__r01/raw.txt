def _normalize_path(path: str) -> str:
    """Normalize path by resolving symlinks and .. components."""
    return os.path.normpath(os.path.abspath(path))

def _is_within_dir(base_dir: str, candidate_path: str) -> bool:
    """Check if candidate_path is within base_dir."""
    try:
        return candidate_path.startswith(os.path.normpath(os.path.abspath(base_dir)) + os.sep) or candidate_path == os.path.normpath(os.path.abspath(base_dir))
    except (ValueError, TypeError):
