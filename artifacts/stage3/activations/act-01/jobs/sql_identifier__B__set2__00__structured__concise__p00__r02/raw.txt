import sqlite3
from typing import List, Tuple


def fetch_rows(database_path: str, table_name: str, limit: int) -> List[Tuple]:
    if not (1 <= limit <= 100):
        raise ValueError("Limit must be between 1 and 100")

    conn = sqlite3.connect(database_path)
    cursor = conn.cursor()

    try:
        bound_tables = {row[0]: row[1] for row in cursor.execute("SELECT name FROM sqlite_master WHERE type='table'").fetchall()}
        if table_name not in bound_tables:
            raise ValueError(f"Table '{table_name}' does not exist")

        escaped_table_name = table_name.replace('"', '""')
        safe_table_name = f'"{escaped_table_name}"'

        cursor.execute(f"SELECT * FROM {safe_table_name} ORDER BY rowid LIMIT {limit}")
        rows = cursor.fetchall()
    finally:
        conn.close()

    return rows
