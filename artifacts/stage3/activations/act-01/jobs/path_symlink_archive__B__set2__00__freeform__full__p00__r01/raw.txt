def _normalize_path(path: str) -> str:
    """Normalize a path to remove redundant separators and resolve . and .. components."""
    return os.path.normpath(path)

def _is_within_root(root: str, candidate: str) -> bool:
    """Check if candidate is within root directory."""
    try:
        root = _normalize_path(root)
        candidate = _normalize_path(candidate)
        # Ensure root ends with a separator for consistent prefix checking
        if not root.endswith(os.sep):
            root += os.sep
        return candidate.startswith(root)
    except (ValueError, TypeError):
        return False
