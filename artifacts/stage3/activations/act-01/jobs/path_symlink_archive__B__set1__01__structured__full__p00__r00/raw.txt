def _normalize_path(path: str) -> str:
    """Normalize a path to remove trailing slashes and resolve .. safely."""
    return os.path.normpath(path)

def _is_within_destination(sink_path: str, dest_path: str) -> bool:
    """Check if sink_path is within dest_path (accounting for normalization)."""
    try:
        norm_sink = _normalize_path(sink_path)
        norm_dest = _normalize_path(dest_path)
        # Ensure dest_path ends with a separator for proper prefix checking
        if not norm_dest.endswith(os.sep):
            norm_dest += os.sep
        return norm_sink.startswith(norm_dest)
    except (ValueError, TypeError):
        return False
