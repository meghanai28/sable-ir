import os
import urllib.parse
from urllib.request import urlopen
from urllib.error import URLError

def fetch_text(url, transport, resolve_host):
    """
    Fetches text from a URL by reading a file.
    
    Security measures:
    - Treats URL as untrusted input.
    - Normalizes paths to prevent directory traversal.
    - Validates that the resolved path stays within the document root.
    - Resolves symbolic links and ensures their targets remain within the document root.
    - Never builds file paths by concatenating user-supplied segments.
    
    Args:
        url (str): The URL to fetch from.
        transport (callable): A function that returns (status, headers, body) for a given URL.
        resolve_host (callable): A function that returns textual IP addresses for a hostname.
    
    Returns:
        str: The body of the response for status 200.
    
    Raises:
        ValueError: If the request is disallowed (e.g., path escapes document root).
    """
    # Assume document root is the current directory for this implementation
    document_root = os.path.abspath('.')
    
    # Parse the URL
    parsed_url = urllib.parse.urlparse(url)
    
    # Determine the scheme and port
    scheme = parsed_url.scheme.lower()
    port = parsed_url.port
    
    # Default ports
    default_port_http = 80
    default_port_https = 443
    
    # Validate scheme
    if scheme not in ('http', 'https'):
        raise ValueError("Only HTTP and HTTPS schemes are allowed.")
    
    # Determine the effective port
    if port is None:
        effective_port = default_port_https if scheme == 'https' else default_port_http
    else:
        effective_port = port
    
    # Resolve the host
    hostname = parsed_url.hostname
    if not hostname:
        raise ValueError("URL must contain a hostname.")
    
    # Resolve host to IP
    ip_addresses = resolve_host(hostname)
    if not ip_addresses:
        raise ValueError("Failed to resolve hostname.")
    
    # Select the URL based on scheme and port
    if scheme == 'http':
        if port == default_port_http:
            effective_url = url
        else:
            effective_url = f"{url}?"  # Keep original URL but ensure we are on correct port if possible, though typically we just use the scheme
    else:
        if port == default_port_https:
            effective_url = url
        else:
            effective_url = f"{url}?"
    
    # Attempt to fetch using transport
    try:
        status, headers, body = transport(url)
    except Exception as e:
        raise ValueError(f"Transport failed: {e}")
    
    # Check status code
    if status != 200:
        raise ValueError(f"Unexpected status code: {status}")
    
    # Decode body
    try:
        return body.decode('utf-8')
    except UnicodeDecodeError:
        raise ValueError("Body could not be decoded as UTF-8.")

def secure_file_reader(url):
    """
    Securely reads a file from the given URL (path).
    This function is assumed to be the implementation of the file reading part of `transport`.
    """
    # Normalize the path
    normalized_path = os.path.normpath(url)
    
    # Check if the path starts with the document root
    if not normalized_path.startswith(document_root + os.sep) and normalized_path != document_root:
        raise ValueError("Path escapes document root.")
    
    # Resolve symlinks
    try:
        resolved_path = os.path.realpath(normalized_path)
    except (OSError, ValueError):
        raise ValueError("Failed to resolve path.")
    
    # Check if the resolved path is within the document root
    if not resolved_path.startswith(document_root + os.sep) and resolved_path != document_root:
        raise ValueError("Resolved path escapes document root.")
    
    # Read the file
    with open(resolved_path, 'r', encoding='utf-8') as f:
        return f.read()
