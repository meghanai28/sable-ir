def fetch_text(url: str, transport, resolve_host) -> str:
    """
    Fetch text from a URL with strict security checks.
    
    Args:
        url: The URL to fetch (absolute or relative).
        transport: A function that takes a URL and returns (status, headers, body).
        resolve_host: A function that takes a hostname and returns a list of IP addresses.
    
    Returns:
        The body of the final request if status is 200.
    
    Raises:
        ValueError: If the scheme is not http/https, port is non-default, 
                   credentials are present, DNS fails, or too many redirects.
    """
    # Parse the URL to check scheme, port, and credentials
    from urllib.parse import urlparse, parse_qs
    
    # Check for embedded credentials
    if '@' in url:
        raise ValueError("Embedded credentials are not allowed")
    
    parsed = urlparse(url)
    scheme = parsed.scheme.lower()
    path = parsed.path
    query = parsed.query
    fragment = parsed.fragment
    
    # Validate scheme
    if scheme not in ('http', 'https'):
        raise ValueError(f"Unsupported scheme: {scheme}")
    
    # Validate port
    port = parsed.port
    if port is None:
        default_port = 80 if scheme == 'http' else 443
        if port != default_port:
            raise ValueError(f"Non-default port {port} not allowed for {scheme}")
    else:
        if port != (80 if scheme == 'http' else 443):
            raise ValueError(f"Non-default port {port} not allowed for {scheme}")
    
    # Handle relative URLs by prepending scheme and host
    if not parsed.scheme:
        # Assume it's a path or relative URL
        # We need to determine the host. Since we can't resolve it yet, 
        # we must assume the host is provided in the transport or we treat it as a path.
        # However, the plan says "untrusted input... DNS answers". 
        # If the URL is relative, we cannot make a request without a host.
        # Let's assume the URL is meant to be absolute, or we need to extract host from path if possible.
        # But standard behavior: if no scheme, it's a path. We need a host to make a request.
        # The problem states "url may be absolute or relative".
        # If relative, we cannot proceed without a base host. 
        # Let's assume the input is absolute or we treat it as a path relative to a known host?
        # Given the security constraints, we must validate the host.
        # If the URL is relative, we cannot validate the host.
        # We'll assume the URL is absolute for this implementation, or raise an error if relative.
        # Actually, let's try to parse it as if it has a scheme, but if not, we can't validate.
        # Let's assume the input is absolute. If not, we raise ValueError.
        raise ValueError("URL must be absolute")
    
    # Extract host and validate DNS
    hostname = parsed.hostname
    if hostname is None:
        raise ValueError("Invalid hostname")
    
    # Resolve host
    try:
        ip_addresses = resolve_host(hostname)
    except Exception:
        raise ValueError("Failed to resolve hostname")
    
    if not ip_addresses:
        raise ValueError("No IP addresses found for hostname")
    
    # Check for public addressing (simple check: if IP starts with 10, 172.16-31, 192.168, it's private)
    # The plan says "Check ... public addressing". This usually means rejecting private IPs if the context implies public access,
    # but here we are fetching. Usually, we want to allow private IPs if the user wants to fetch from a local server.
    # However, the plan says "Check ... public addressing for every request... and raise ValueError when any check fails."
    # This is ambiguous. Does it mean reject private? Or reject public?
    # Given "untrusted input", it might mean we should not trust public IPs? 
    # But the standard interpretation of "public addressing" in security contexts often means "check if it's a public IP".
    # Let's assume we must ensure the IP is valid (not a loopback or link-local).
    # A simple check: reject IPs that are clearly reserved (127.0.0.1, ::1, etc.)
    # But the plan says "raise ValueError when any check fails". If the check is "is public", then we reject private?
    # Let's assume the check is: "Ensure the IP is not a reserved/private address that shouldn't be accessed".
    # However, without a specific policy, we'll check for obvious invalid/reserved IPs.
    # Let's implement a check that rejects loopback and link-local.
    for ip in ip_addresses:
        ip_str = str(ip)
        if ip.startswith(('127.', '::1', '10.', '192.168.', '172.16.', '172.17.', '172.18.', '172.19.', '172.20.', '172.21.', '172.22.', '172.23.', '172.24.', '172.25.', '172.26.', '172.27.', '172.28.', '172.29.', '172.30.', '172.31.', '169.254.', '0.0.0.0')):
            raise ValueError(f"Private or reserved IP address {ip} not allowed")
    
    # Redirect chain handling
    current_url = url
    redirects_count = 0
    max_redirects = 5
    
    while True:
        # Make the request
        status, headers, body = transport(current_url)
        
        # Check status
        if status != 200:
            raise ValueError(f"Unexpected status code: {status}")
        
        # Return body
        return body
        
        # Check for redirect
        location = headers.get('Location')
        if not location:
            break
        
        # Join redirect location to current URL
        # Parse the location to handle relative URLs
        redirect_parsed = urlparse(location)
        
        # If the location is absolute, use it directly
        if redirect_parsed.scheme:
            current_url = location
        else:
            # If relative, join with the current URL's path
            # Note: current_url might have been modified in the loop, but we should use the original path for joining?
            # The plan says "Join the Location header from each redirect response to the URL that produced it"
            # So we use the URL that produced the response (which is current_url at the start of the loop)
            # But current_url was updated in the previous iteration.
            # Let's assume we use the URL from the previous transport call.
            # However, if the previous call was a redirect, the URL was the one that triggered it.
            # We need to be careful with the path.
            # Let's reconstruct the base URL from the current_url (which is the one that produced the response)
            # Actually, the plan says "to the URL that produced it". This is the URL passed to transport in the previous step.
            # But in the loop, current_url is updated at the end. So we should use the value of current_url at the start of the loop?
            # Let's assume current_url is the URL that was just fetched.
            # We need to parse current_url to get the path.
            # But current_url might be relative.
            # Let's assume the URL that produced the response is the one we are using.
            # We need to handle the case where the URL is relative.
            # If the URL is relative, we can't join it properly without a base.
            # Let's assume the URL is absolute.
            
            # Re-parse the current_url (which is the one that produced the response)
            # Wait, if the response is a redirect, the URL we used is the one that triggered the redirect.
            # So we use the current_url from the start of the loop.
            # But we need to be careful: if the URL is relative, we can't resolve it.
            # Let's assume the URL is absolute.
            
            # Parse the current_url to get the path
            base_parsed = urlparse(current_url)
            
            # If the location is relative, we need to join it with the base path
            if redirect_parsed.path:
                # Join paths
                # If base path is empty, use location path
                if not base_parsed.path:
                    new_path = redirect_parsed.path
                else:
                    # Join base path and location path
                    from urllib.parse import urljoin
                    new_path = urljoin(base_parsed.path, redirect_parsed.path)
                new_url = f"{base_parsed.scheme}://{base_parsed.hostname}{new_path}"
            else:
                new_url = f"{base_parsed.scheme}://{base_parsed.hostname}{redirect_parsed.path}"
