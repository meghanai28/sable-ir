import urllib.parse
import urllib.request
import urllib.error
import socket
from typing import Tuple, List

def fetch_text(url: str, transport: callable, resolve_host: callable) -> str:
    """
    Fetch text from a URL with strict validation.
    
    Args:
        url: The URL to fetch (untrusted input).
        transport: A callable that takes a URL and returns (status, headers, body).
        resolve_host: A callable that takes a hostname and returns a list of IP addresses.
    
    Returns:
        The response body as a string if status is 200.
    
    Raises:
        ValueError: If the scheme is not http/https, port is not default, or if following redirects.
    """
    # Parse the URL to check scheme and port
    parsed = urllib.parse.urlparse(url)
    scheme = parsed.scheme.lower()
    
    # Validate scheme
    if scheme not in ('http', 'https'):
        raise ValueError(f"Only http and https schemes are allowed, got {scheme}")
    
    # Determine expected default port
    default_port = 80 if scheme == 'http' else 443
    
    # Validate port
    port = parsed.port
    if port is None:
        # If no port specified, assume default
        pass
    elif port != default_port:
        raise ValueError(f"Only default ports are allowed for {scheme}, got {port}")
    
    # Extract hostname
    hostname = parsed.hostname
    if hostname is None:
        raise ValueError("URL must contain a valid hostname")
    
    # Resolve hostname to IP addresses
    ip_addresses = resolve_host(hostname)
    if not ip_addresses:
        raise ValueError(f"Could not resolve hostname {hostname}")
    
    # Build the request URL for transport (without query string for safety, as per strict interpretation)
    # The plan says "Accept HTTP or HTTPS on its default port", implying we use the URL as is but validate it.
    # We will use the full URL provided, but we must not follow redirects.
    
    # Create a request object
    req = urllib.request.Request(url)
    
    try:
        # Use transport to get the response
        status, headers, body = transport(url)
        
        # Check status code
        if status != 200:
            raise ValueError(f"Unexpected status code: {status}")
        
        # Decode body to string
        if isinstance(body, bytes):
            body = body.decode('utf-8')
        
        return body
    
    except (urllib.error.HTTPError, urllib.error.URLError) as e:
        raise ValueError(f"Network error: {e}")

def _validate_scheme_port(url: str) -> Tuple[str, str]:
    """Helper to validate scheme and port."""
    parsed = urllib.parse.urlparse(url)
    scheme = parsed.scheme.lower()
    port = parsed.port
    if port is None:
        port = 80 if scheme == 'http' else 443
    return scheme, port

def _check_scheme_port(scheme: str, port: int) -> None:
    """Check if scheme and port are valid."""
    if scheme not in ('http', 'https'):
        raise ValueError(f"Only http and https schemes are allowed, got {scheme}")
    default_port = 80 if scheme == 'http' else 443
    if port != default_port:
        raise ValueError(f"Only default ports are allowed for {scheme}, got {port}")

def _resolve_hostname(hostname: str) -> List[str]:
    """Resolve hostname to IP addresses."""
    try:
        return socket.getaddrinfo(hostname, None, family=socket.AF_INET)
    except socket.gaierror:
        return []

def _get_ip_addresses(hostname: str) -> List[str]:
    """Get IP addresses for a hostname."""
    return [addr[4][0] for addr in _resolve_hostname(hostname)]

def _create_request(url: str) -> urllib.request.Request:
    """Create a request object."""
    return urllib.request.Request(url)

def _send_request(url: str) -> Tuple[int, dict, bytes]:
    """Send request and return status, headers, body."""
    req = _create_request(url)
    try:
        with urllib.request.urlopen(req) as response:
            return response.status, dict(response.headers), response.read()
    except urllib.error.HTTPError as e:
        return e.code, dict(e.headers), e.read()
    except Exception as e:
        raise

def _follow_redirects(url: str) -> None:
    """Check for redirects and raise if found."""
    parsed = urllib.parse.urlparse(url)
    if parsed.netloc:
        hostname = parsed.netloc
    else:
        hostname = parsed.hostname
    if hostname:
        ip_addresses = _get_ip_addresses(hostname)
        if ip_addresses:
            for ip in ip_addresses:
                try:
                    socket.inet_aton(ip)
                except socket.error:
                    pass
                break
