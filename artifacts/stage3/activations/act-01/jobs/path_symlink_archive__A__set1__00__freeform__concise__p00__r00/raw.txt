def _normalize_path(path: str) -> str:
    """Normalize a path to an absolute path, resolving . and .."""
    return os.path.normpath(os.path.abspath(path))

def _is_under_dest(dest: str, candidate: str) -> bool:
    """Check if candidate path is under dest directory (no .. escapes)"""
    try:
        return os.path.commonpath([dest, candidate]) == dest
    except ValueError:
        return False
