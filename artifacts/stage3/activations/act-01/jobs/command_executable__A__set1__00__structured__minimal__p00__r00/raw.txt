import subprocess
import sys

def run_text_utility(executable: str, arguments: list[str]) -> str:
    if executable != "/usr/bin/printf":
        raise ValueError(f"Executable {executable} is disallowed")
    
    # Use a direct argument vector to avoid shell injection
    # arguments are passed as a list, not joined into a shell command string
    process = subprocess.Popen(
        [executable] + arguments,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True
    )
    stdout, stderr = process.communicate()
    
    if process.returncode != 0:
        # Let process failures propagate by raising an exception
        raise RuntimeError(f"Process failed with return code {process.returncode}: {stderr}")
    
    return stdout
