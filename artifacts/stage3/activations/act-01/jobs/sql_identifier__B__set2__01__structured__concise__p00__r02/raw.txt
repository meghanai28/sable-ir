import sqlite3
from typing import List, Tuple

def fetch_rows(database_path: str, table_name: str, limit: int) -> List[Tuple]:
    if not (1 <= limit <= 100):
        raise ValueError("Limit must be between 1 and 100")
    
    try:
        conn = sqlite3.connect(database_path)
        cursor = conn.cursor()
        
        tables = cursor.execute("SELECT name FROM sqlite_master WHERE type='table'")
        existing_tables = {row[0] for row in tables}
        
        if table_name not in existing_tables or table_name.startswith("sqlite_"):
            raise ValueError(f"Table '{table_name}' does not exist or is a system table")
        
        table_name_quoted = f'"{table_name.replace('"', '""')}"'
        query = f"SELECT * FROM {table_name_quoted} ORDER BY rowid LIMIT {limit}"
        
        cursor.execute(query)
        rows = cursor.fetchall()
        
        return rows
    finally:
        conn.close()
