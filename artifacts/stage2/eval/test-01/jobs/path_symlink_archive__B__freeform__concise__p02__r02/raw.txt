import os
import tarfile

def extract_tar_to_path(tar_path: str, dest_path: str) -> bool:
    """
    Extract a tar archive to a destination path with strict security validation.
    
    Args:
        tar_path: Path to the tar archive file.
        dest_path: Destination directory for extraction.
        
    Returns:
        True if extraction completes successfully without writing outside dest_path.
        False if validation or extraction fails.
    """
    try:
        # Normalize paths to handle trailing slashes and ensure consistency
        tar_path = os.path.normpath(tar_path)
        dest_path = os.path.normpath(dest_path)
        
        # Resolve the actual directory for dest_path to handle symlinks in the path itself
        # but keep the check strict against .. escapes in the resolved path
        dest_path_resolved = os.path.realpath(dest_path)
        
        # Security check: Ensure dest_path_resolved is within the current working directory's root
        # We check if dest_path_resolved starts with the current directory to prevent absolute path escapes
        # However, a more robust check is to ensure it doesn't contain '..' after normalization and is not absolute if not intended
        # The requirement says "raise an error if it escapes the root". We interpret "root" as the filesystem root.
        # We must ensure dest_path_resolved does not start with '/' unless it's the desired root (which is risky)
        # Actually, the safest approach per "escapes the root" is to check if the path is absolute and points to a location outside a safe base,
        # but since no base is provided, we assume the user's intent is to extract relative to the current dir or a specific dest.
        # The critical check is: does the resolved path contain '..' that goes above the intended extraction root?
        # We will enforce that dest_path_resolved must not be an absolute path starting with '/' unless we trust the environment,
        # but the plan says "raise an error if it escapes the root". This usually means preventing absolute paths or paths going above /.
        # To be safe, we ensure dest_path_resolved is not an absolute path if it contains '..', or simply check that it's not absolute
        # and doesn't start with '/'.
        if dest_path_resolved.startswith('/') or dest_path_resolved == '/':
            return False
            
        # Additional check: ensure no '..' in the normalized path that could escape
        # Since we normalized, we just check if the resolved path is safe.
        # We will use a helper to check if a path is safe (doesn't escape root).
        if not _is_path_safe(dest_path_resolved):
            return False
            
        # Create the destination directory if it doesn't exist
        os.makedirs(dest_path_resolved, exist_ok=True)
        
        # Open the tar file
        with tarfile.open(tar_path, 'r:*') as tar:
            # Get list of all members
            members = tar.getmembers()
            
            # Validate all members before extraction
            for member in members:
                # Check if the member's name starts with '..' (after normalization)
                # We need to resolve the full path relative to dest_path_resolved
                member_name = member.name
                if member_name.startswith('..'):
                    return False
                
                # Calculate the full target path
                full_target = os.path.join(dest_path_resolved, member_name)
                
                # Resolve the target to check for .. escapes
                # We must resolve symlinks and hard links to ensure they stay inside dest_path
                try:
                    resolved_target = os.path.realpath(full_target)
                except (OSError, ValueError):
                    return False
                    
                # Security check: Ensure resolved_target is within dest_path_resolved
                if not _is_path_safe(resolved_target) or not resolved_target.startswith(dest_path_resolved + os.sep) and resolved_target != dest_path_resolved:
                    # Wait, if dest_path_resolved is a directory, the target must start with dest_path_resolved + os.sep
                    # But if the member is '.', it should be dest_path_resolved
                    # Let's refine: the target must be inside dest_path_resolved
                    if not (resolved_target.startswith(dest_path_resolved + os.sep) or resolved_target == dest_path_resolved):
                        return False
                        
                # Check for symbolic links and hard links
                if member.issym() or member.islnk():
                    # Resolve the link target
                    link_target = member.linkname
                    # Normalize the link target to remove ..
                    normalized_link = os.path.normpath(link_target)
                    if normalized_link.startswith('..'):
                        return False
                    
                    # Resolve the full link target path
                    full_link_target = os.path.join(dest_path_resolved, normalized_link)
                    resolved_link_target = os.path.realpath(full_link_target)
                    
                    # Check if the resolved link target is safe and inside dest_path
                    if not _is_path_safe(resolved_link_target):
                        return False
                    if not (resolved_link_target.startswith(dest_path_resolved + os.sep) or resolved_link_target == dest_path_resolved):
                        return False
                        
            # Perform extraction
            try:
                tar.extractall(path=dest_path_resolved)
                return True
            except Exception:
                return False
                
    except Exception:
        return False
