def _normalize_path(path: str) -> str:
    """Normalize a path by resolving .. and . segments without resolving symlinks."""
    return os.path.normpath(path)


def _is_safe_relative_to(root: str, path: str) -> bool:
    """Check if path is safely within root without following symlinks."""
    try:
        norm_root = _normalize_path(root)
        norm_path = _normalize_path(path)
        # Ensure norm_root ends with a separator for proper prefix checking
        if not norm_root.endswith(os.sep):
            norm_root += os.sep
        return norm_path.startswith(norm_root)
    except (ValueError, TypeError):
        return False


def _is_safe_target(target: str, dest_root: str) -> bool:
    """Check if a symlink/hardlink target is safely within dest_root."""
    try:
        norm_target = _normalize_path(target)
        return _is_safe_relative_to(dest_root, norm_target)
    except (ValueError, TypeError):
        return False
