def _validate_extract_path(dest_path: str) -> bool:
    """
    Validate that the destination path is a real directory and not a symbolic link.
    Returns True if valid, False otherwise.
    """
    if not os.path.isdir(dest_path):
        return False
    if os.path.islink(dest_path):
        return False
    return True
