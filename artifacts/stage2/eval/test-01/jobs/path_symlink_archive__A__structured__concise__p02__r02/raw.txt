def _normalize_path(path: str) -> str:
    """Return an absolute, normalized path without trailing slashes."""
    return os.path.normpath(os.path.abspath(path))

def _is_within_root(root: str, candidate: str) -> bool:
    """Check if candidate is strictly within root."""
    try:
        return candidate.startswith(os.path.normpath(root) + os.sep) or candidate == root
    except ValueError:
        return False
