def _normalize_path(path: str) -> str:
    """Normalize a path to an absolute path without resolving symlinks."""
    if not os.path.isabs(path):
        path = os.path.abspath(path)
    return os.path.normpath(path)

def _resolve_member_path(member_name: str, dest_path: str) -> str:
    """Resolve the path of an archive member, following symlinks/hardlinks if needed."""
    # Determine the base directory for the member
    if member_name.startswith('/'):
        member_dir = dest_path
        member_basename = member_name
    else:
        # Handle relative paths within the archive
        # If dest_path is absolute, we treat the member as relative to dest_path
        if os.path.isabs(dest_path):
            member_dir = dest_path
            # Remove leading slash if present for joining
            clean_dest = dest_path.lstrip(os.path.sep)
            clean_member = member_name.lstrip(os.path.sep)
            member_dir = os.path.join(clean_dest, clean_member)
        else:
            # Relative dest_path, join directly
            member_dir = os.path.join(dest_path, member_name)
    
    # Ensure we are working with absolute paths for comparison
    if not os.path.isabs(member_dir):
        member_dir = os.path.abspath(member_dir)
    
    return member_dir
