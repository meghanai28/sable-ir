def _normalize_url(url):
    """Normalize URL, resolving relative parts and ensuring absolute form."""
    parsed = urllib.parse.urlparse(url)
    if not parsed.scheme:
        # Relative URL
        base = urllib.parse.urlparse(urllib.request.urljoin('file://' + __file__, ''))
        # For relative URLs, we treat the current directory as the base.
        # However, the spec implies a directory-based containment check.
        # We will assume the base is the current working directory's root for simplicity
        # or construct a base from the first absolute URL seen.
        # Since we start with a potentially relative URL, we need a base.
        # Let's assume the "current effective directory" is the directory of the script
        # or the directory of the first absolute URL encountered.
        # To be safe and strictly follow "escapes current effective directory",
        # we need a base. Let's use the script's directory as the default base if no absolute URL is given.
        base = urllib.parse.urlparse(f"file://{sys.path[0].rstrip('/')}/")
        normalized = urllib.parse.urljoin(base.getpath(), parsed.path)
        if normalized.startswith('./') or normalized.startswith('../'):
            normalized = normalized[2:]
        # Handle query and fragment
        if parsed.query:
            normalized = normalized + '?' + parsed.query
        if parsed.fragment:
            normalized = normalized + '#' + parsed.fragment
        return urllib.parse.urljoin('file://' + base.getpath(), normalized)
    else:
        return url

def _validate_location(current_url, location):
    """Validate that the location does not escape the current directory and matches scheme/port."""
    parsed_current = urllib.parse.urlparse(current_url)
    parsed_location = urllib.parse.urlparse(location)
    
    # Check scheme
    if parsed_location.scheme != parsed_current.scheme:
        raise ValueError("Scheme mismatch")
    
    # Check port
    if parsed_current.port and parsed_location.port != parsed_current.port:
        raise ValueError("Port mismatch")
    
    # Resolve location against current URL
    # If current is absolute, join; if relative, treat as path relative to current path
    if parsed_current.scheme:
        new_url = urllib.parse.urljoin(current_url, location)
    else:
        # If current is relative, we need a base.
        # Assuming the current URL's path is the base.
        base_path = parsed_current.path
        if base_path.endswith('/'):
            new_url = urllib.parse.urljoin('file://' + base_path, location)
        else:
            new_url = urllib.parse.urljoin('file://' + base_path + '/', location)
    
    return new_url

def _get_effective_directory(url):
    """Extract the effective directory from the URL for containment checks."""
    parsed = urllib.parse.urlparse(url)
    if parsed.scheme:
        return parsed.path.rstrip('/')
    else:
        # For relative, assume current dir
        return '.'

def _is_within_directory(url, base_dir):
    """Check if url is within base_dir."""
    # Normalize both
    normalized_url = urllib.parse.urljoin(base_dir, url)
    # Remove leading slashes
    normalized_url = normalized_url.lstrip('/')
    return normalized_url.startswith(base_dir.rstrip('/')) or normalized_url == base_dir.rstrip('/')
