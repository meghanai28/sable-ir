def _validate_scheme_port(url: str, scheme: str, port: int) -> bool:
    """Ensure scheme and port are allowed."""
    allowed_schemes = {'http', 'https'}
    allowed_ports = {80, 443}
    
    if scheme not in allowed_schemes:
        return False
    
    # Parse the port from the URL
    parsed = urllib.parse.urlparse(url)
    if parsed.port is None:
        if scheme == 'http':
            return False
        elif scheme == 'https':
            return False
    else:
        if parsed.port not in allowed_ports:
            return False
    
    return True

def _resolve_location(current_url: str, location: str) -> str:
    """Resolve a Location header against the current URL."""
    return urllib.parse.urljoin(current_url, location)

def _get_effective_directory(url: str) -> str:
    """Get the effective directory (path without trailing slash) from the URL."""
    parsed = urllib.parse.urlparse(url)
    path = parsed.path.rstrip('/')
    if not path:
        return '/'
    return path

def _validate_location_safety(current_url: str, location: str) -> bool:
    """Validate that the location does not escape the current directory."""
    # Resolve the location
    resolved = _resolve_location(current_url, location)
    
    # Parse the resolved URL
    parsed_resolved = urllib.parse.urlparse(resolved)
    
    # Check scheme
    if parsed_resolved.scheme:
        return False
    
    # Check netloc
    if parsed_resolved.netloc:
        return False
    
    # Check path
    base_path = _get_effective_directory(current_url)
    resolved_path = parsed_resolved.path.rstrip('/')
    
    # If resolved path is absolute (starts with /), it's unsafe unless it matches base
    if resolved_path.startswith('/'):
        if not resolved_path.startswith(base_path + '/') and base_path:
            return False
        # If base is root, then it's fine
        if not base_path:
            return True
    
    # Check for .. segments
    # We need to ensure that no .. goes beyond the base path
    # Split the resolved path into segments
    if '..' in resolved_path:
        # Check if the .. is within the base path
        # We can do this by checking the depth
        base_segments = base_path.split('/') if base_path else []
        resolved_segments = resolved_path.split('/')
        
        # Find the first ..
        first_dotdot_idx = -1
        for i, seg in enumerate(resolved_segments):
            if seg == '..':
                first_dotdot_idx = i
                break
        
        if first_dotdot_idx != -1:
            # Check if the segment before the .. is within the base path
            # The segment before .. should be part of the base path
            if first_dotdot_idx >= len(base_segments):
                return False
            
            # Check if the .. is followed by something that would go beyond base
            # Actually, the simplest check is: the resolved path must be a subpath of the base path
            # If the resolved path contains .., it must not escape the base
            
            # Let's check if the resolved path is within the base path
            # We can do this by checking if the resolved path starts with the base path
            if not resolved_path.startswith(base_path + '/') and base_path:
                return False
    
    return True
