def _is_public_dns(hostname: str) -> bool:
    """
    Basic heuristic to check if a hostname looks like a public DNS entry.
    This is a simplified check; in a real security context, this would be
    replaced with a proper DNS resolution check or a whitelist.
    """
    if not hostname:
        return False
    # Reject internal IP ranges (127.0.0.0/8, 10.0.0.0/8, 192.168.0.0/16, etc.)
    # and localhost variations.
    ip_pattern = r'^(\d{1,3}\.){3}\d{1,3}$'
    if re.match(ip_pattern, hostname):
        parts = hostname.split('.')
        if len(parts) == 4:
            octets = [int(p) for p in parts]
            if (octets[0] == 127) or (octets[0] == 10) or (octets[0] == 192 and octets[1] == 168) or (octets[0] == 172 and 16 <= octets[1] <= 31) or (octets[0] == 169 or octets[0] == 126):
                return False
    # Reject obvious internal domains (example: local, intranet, etc.)
    internal_keywords = ['local', 'intranet', 'localhost', 'test', 'example']
    if any(kw in hostname.lower() for kw in internal_keywords):
        return False
    return True

def _validate_location(location: str, current_url: str) -> None:
    """
    Validate the redirect location against the current URL.
    Raises ValueError if:
    - Scheme is different (not allowed)
    - Port is different (not allowed)
    - Credentials are missing (not allowed)
    - Hostname is not a public DNS entry
    """
    # Parse current URL
    current_parsed = urllib.parse.urlparse(current_url)
    
    # Parse location URL
    # Note: location might be absolute or relative
    if location.startswith(('http://', 'https://')):
        location_parsed = urllib.parse.urlparse(location)
    else:
        # Relative URL
        base = urllib.parse.urlparse(current_url)
        location_parsed = urllib.parse.urlparse(urllib.parse.urljoin(base.geturl(), location))
    
    # Check scheme
    if location_parsed.scheme != current_parsed.scheme:
        raise ValueError("Redirect scheme mismatch")
    
    # Check port
    if location_parsed.port != current_parsed.port:
        raise ValueError("Redirect port mismatch")
    
    # Check credentials
    # If current URL has credentials, location must have them too
    if current_parsed.username or current_parsed.password:
        if not (location_parsed.username and location_parsed.password):
            raise ValueError("Missing credentials in redirect")
    
    # Check DNS
    host = location_parsed.hostname
    if not _is_public_dns(host):
        raise ValueError("Non-public DNS entry in redirect")
