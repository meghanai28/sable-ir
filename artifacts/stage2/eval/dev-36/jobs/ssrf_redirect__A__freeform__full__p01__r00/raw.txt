def fetch_text(url, transport, resolve_host):
    """
    Fetch text from a URL with strict redirect and scheme/domain validation.
    
    Args:
        url: Absolute or relative URL string.
        transport: Function(url) -> (status, headers, body) tuple.
        resolve_host: Function(hostname) -> list of textual IP addresses.
    
    Returns:
        str: The body of the response with status 200.
    
    Raises:
        ValueError: If status is not 200, scheme/domain mismatch, or redirect rules violated.
    """
    from urllib.parse import urlparse, urljoin
    
    # Parse initial URL
    parsed_url = urlparse(url)
    original_scheme = parsed_url.scheme
    original_domain = parsed_url.netloc
    original_path = parsed_url.path
    
    # Determine scheme and port
    if original_scheme.lower() == 'http':
        scheme = 'http'
        port = 80
    elif original_scheme.lower() == 'https':
        scheme = 'https'
        port = 443
    else:
        raise ValueError("Only HTTP and HTTPS schemes are allowed.")
    
    # Helper to check if a URL is within the same scheme and domain
    def is_same_origin(target_url):
        try:
            target_parsed = urlparse(target_url)
            if target_parsed.scheme.lower() != original_scheme.lower():
                return False
            if target_parsed.netloc.lower() != original_domain.lower():
                return False
            return True
        except Exception:
            return False
    
    # Helper to resolve hostname to IP
    def resolve_and_validate_hostname(hostname):
        if not hostname:
            raise ValueError("Empty hostname")
        ips = resolve_host(hostname)
        if not ips:
            raise ValueError("No IP addresses resolved")
        # Check if at least one IP is textual (all should be per spec, but safety check)
        if not all(isinstance(ip, str) for ip in ips):
            raise ValueError("Resolved IPs must be textual")
        return ips[0]  # Return first valid IP
    
    # Function to make the actual HTTP request
    def make_request(current_url, current_scheme, current_domain, current_path, current_port):
        # Build the full URL for transport
        if not current_url.startswith('http://') and not current_url.startswith('https://'):
            # It's relative, need to join with base
            base = f"{scheme}://{current_domain}:{current_port}"
            full_url = urljoin(base, current_url)
        else:
            full_url = current_url
        
        # Resolve the final destination hostname
        # We need to resolve the hostname from the full URL
        # If it's absolute, use netloc; if relative, we already resolved it in the join logic above
        # But wait, the spec says "Resolve the target's hostname". 
        # If the URL is absolute, we resolve its netloc.
        # If relative, we resolve the netloc of the base we used.
        
        # Actually, let's re-parse the full_url to get the netloc for resolution
        # But if it's absolute, the netloc is the target. If relative, the netloc of the base is the target.
        
        # Re-parse full_url
        target_parsed = urlparse(full_url)
        target_netloc = target_parsed.netloc
        
        # If the URL was absolute, target_netloc is the target.
        # If the URL was relative, we joined with a base, so target_netloc is the base's netloc.
        # This is correct.
        
        # Resolve the hostname
        try:
            resolved_ips = resolve_host(target_netloc)
        except Exception:
            raise ValueError("Failed to resolve hostname")
        
        if not resolved_ips:
            raise ValueError("No IP addresses resolved")
        
        # Check if the resolved IP is textual
        if not all(isinstance(ip, str) for ip in resolved_ips):
            raise ValueError("Resolved IPs must be textual")
        
        # Check if the scheme of the full_url matches the resolved scheme (it should, but verify)
        # Actually, the scheme is already determined by the URL string.
        # The check is that the resolved destination remains within the same scheme and domain.
        # Since we are resolving the hostname from the URL we are about to fetch, 
        # we just need to ensure that the URL we are fetching has the correct scheme and domain.
        # The spec says: "raise ValueError unless the final destination remains within the same scheme and domain as the requested url."
        # The "final destination" here is the URL we are about to fetch.
        # We already checked the scheme and domain of the URL we are about to fetch against the original URL.
        # So we just need to make sure the URL we are about to fetch is valid.
        
        # Now make the request
        status, headers, body = transport(full_url)
        
        # Check status
        if status != 200:
            raise ValueError(f"Status code {status} is not 200")
        
        # Check if there is a Location header
        location = headers.get('Location')
        if location:
            # Parse the Location header
            # If it's absolute, it's an absolute URL.
            # If it's relative, it's relative to the current URL.
            if location.startswith(('http://', 'https://')):
                # It's an absolute URL
                # Check if it's within the same scheme and domain
                if not is_same_origin(location):
                    raise ValueError("Redirect to different scheme or domain")
            else:
                # It's relative, join with current URL
                new_url = urljoin(full_url, location)
                # Check if it's within the same scheme and domain
                if not is_same_origin(new_url):
                    raise ValueError("Redirect to different scheme or domain")
