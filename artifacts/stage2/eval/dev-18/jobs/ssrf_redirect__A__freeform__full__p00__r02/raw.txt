import socket
import urllib.parse
from urllib.request import urlopen
from urllib.error import URLError

def fetch_text(url: str, transport, resolve_host) -> str:
    # Parse the URL to extract the scheme, netloc, and path
    parsed = urllib.parse.urlparse(url)
    
    # Validate scheme and port
    scheme = parsed.scheme.lower()
    if scheme not in ('http', 'https'):
        raise ValueError("Only HTTP and HTTPS schemes are allowed")
    
    # Determine the port based on scheme
    if scheme == 'http':
        default_port = 80
    else:
        default_port = 443
    
    # Extract the host
    hostname = parsed.netloc
    if not hostname:
        raise ValueError("Invalid URL: missing host")
    
    # Remove port from hostname if present for resolution
    if ':' in hostname:
        hostname = hostname.split(':')[0]
    
    # Resolve the host
    ip_addresses = resolve_host(hostname)
    if not ip_addresses:
        raise ValueError(f"No IP address found for host: {hostname}")
    
    # Use the first resolved IP address
    resolved_host = ip_addresses[0]
    
    # Construct the full URL for the request
    # Ensure the path/query/fragment are preserved correctly
    full_url = f"{scheme}://{resolved_host}{parsed.path}"
    if parsed.query:
        full_url += f"?{parsed.query}"
    if parsed.fragment:
        full_url += f"#{parsed.fragment}"
    
    # Make the request using the provided transport
    try:
        response = transport(full_url)
    except Exception as e:
        raise ValueError(f"Request failed: {str(e)}")
    
    status, headers, body = response
    
    # Check for disallowed status codes (not 200)
    if status != 200:
        raise ValueError(f"Disallowed status code: {status}")
    
    # Return the body
    if isinstance(body, bytes):
        return body.decode('utf-8')
    else:
        return str(body)
