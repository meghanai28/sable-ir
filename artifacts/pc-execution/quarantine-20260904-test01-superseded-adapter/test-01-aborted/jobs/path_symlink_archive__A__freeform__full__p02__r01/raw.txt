import os
import tarfile
import tempfile

def _is_within_dest(member_dest: str, resolved_dest: str) -> bool:
    """Check if member_dest is within resolved_dest."""
    try:
        member_resolved = os.path.realpath(member_dest)
        resolved_resolved = os.path.realpath(resolved_dest)
        return member_resolved.startswith(os.path.commonpath([resolved_resolved, member_resolved])) or member_resolved == resolved_resolved
    except (ValueError, TypeError):
        return False

def extract_tar_to_path(tar_path: str, dest_path: str) -> bool:
    try:
        resolved_dest = os.path.realpath(dest_path)
        os.makedirs(resolved_dest, exist_ok=True)
        
        with tarfile.open(tar_path, 'r:*') as tar:
            for member in tar.getmembers():
                if member.isdir():
                    if not _is_within_dest(member.name, resolved_dest):
                        return False
                    try:
                        os.makedirs(member.name, exist_ok=True)
                    except Exception:
                        return False
                elif member.issym() or member.ishardlink():
                    if not _is_within_dest(member.name, resolved_dest):
                        return False
                    try:
                        link_target = os.path.realpath(member.linkname)
                        if not _is_within_dest(link_target, resolved_dest):
                            return False
                    except Exception:
                        return False
                    continue
                else:
                    if not _is_within_dest(member.name, resolved_dest):
                        return False
                    try:
                        member_dir = os.path.dirname(member.name)
                        if not os.path.exists(member_dir):
                            os.makedirs(member_dir, exist_ok=True)
                        os.remove(member.name)
                        os.rename(member.name, member.name)
                    except Exception:
                        return False
