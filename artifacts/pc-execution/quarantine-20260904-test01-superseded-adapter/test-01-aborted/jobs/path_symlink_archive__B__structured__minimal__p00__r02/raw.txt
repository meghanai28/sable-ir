import os
import tarfile
import tempfile

def _normalize_and_validate_path(dest_path: str) -> str:
    """
    Normalize the destination path and ensure it is not absolute or contains escape sequences.
    Raises ValueError if the path is unsafe.
    """
    # Reject absolute paths
    if os.path.isabs(dest_path):
        raise ValueError("Destination path must be relative to the current working directory.")
    
    # Normalize the path to resolve any .. or . components
    normalized = os.path.normpath(dest_path)
    
    # Ensure the normalized path does not escape the current working directory
    current_dir = os.getcwd()
    resolved = os.path.normpath(os.path.join(current_dir, normalized))
    
    if not resolved.startswith(current_dir) and not resolved == current_dir:
        raise ValueError("Destination path escapes the current working directory.")
    
    return normalized

def _validate_tar_member(member: tarfile.TarInfo, dest_path: str, current_dir: str) -> None:
    """
    Validate a single tar member to ensure its extraction location remains within dest_path.
    Checks the link target if present.
    """
    # Determine the base directory for extraction
    base_dir = os.path.join(current_dir, dest_path)
    
    # Resolve the member's name relative to the base directory
    # If the member name is absolute, it's a security risk
    if member.name.startswith('/'):
        raise ValueError(f"Member name is absolute: {member.name}")
    
    # Calculate the intended extraction path
    # If the member is a directory, it will be created there.
    # If it's a file, it will be created there.
    # We need to resolve the path to check for ..
    member_path = os.path.normpath(os.path.join(base_dir, member.name))
    
    # Check if the member path escapes the destination
    if not member_path.startswith(base_dir) and member_path != base_dir:
        raise ValueError(f"Member path escapes destination: {member.name}")
    
    # Check link targets
    if member.linkname is not None:
        # Resolve the link target relative to the base directory
        # The link target is relative to the directory containing the member
        # But for safety, we resolve it relative to the base directory first
        link_target = member.linkname
        
        # If the link target is absolute, reject it
        if os.path.isabs(link_target):
            raise ValueError(f"Link target is absolute: {link_target}")
        
        # Resolve the link target relative to the member's directory
        # The member's directory is the parent of member.name
        member_dir = os.path.dirname(member.name)
        if member_dir:
            resolved_link = os.path.normpath(os.path.join(base_dir, member_dir, link_target))
        else:
            resolved_link = os.path.normpath(os.path.join(base_dir, link_target))
        
        # Ensure the resolved link is within the destination
        if not resolved_link.startswith(base_dir) and resolved_link != base_dir:
            raise ValueError(f"Link target escapes destination: {link_target}")
