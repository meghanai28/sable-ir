def _normalize_and_validate_dest(dest_path: str) -> str:
    """
    Normalize the destination path and check for escape attempts.
    Returns the absolute, normalized path if safe, raises ValueError otherwise.
    """
    # Resolve to absolute path
    abs_dest = os.path.abspath(dest_path)
    
    # Ensure the directory exists
    os.makedirs(abs_dest, exist_ok=True)
    
    # Check if the absolute path escapes the intended root (if any) or is just safe
    # The requirement implies we treat dest_path as the root. We must ensure no .. escapes it.
    # Since we use os.path.abspath, .. is resolved. We need to verify that the final
    # absolute path is exactly what we expect and doesn't contain unexpected .. in its components
    # relative to the starting point, but the safest check is to ensure the resolved path
    # doesn't go above the starting point if the starting point was meant to be a directory.
    # However, the spec says "normalize dest_path and raise an error if it escapes the resolved destination root".
    # This implies we should treat the normalized path as the root.
    # We just need to make sure the path is absolute and doesn't contain .. that would resolve outside.
    
    # A simple robust check: if the absolute path starts with the original dest_path (normalized)
    # or if the original dest_path was absolute, we just ensure we don't go above it.
    # Actually, the safest interpretation for "escapes the resolved destination root" is:
    # If dest_path is relative, normalize it. If it contains .., resolve it. If the result
    # is outside the "root" (which is typically the current dir or a specified base), fail.
    # But here, dest_path IS the root. So we just need to ensure we don't create files outside
    # of dest_path. The extraction logic will handle this, but we must validate the path itself.
    
    # Let's assume the "root" is the directory containing dest_path if dest_path is a file-like name,
    # or dest_path itself if it's a directory.
    # The spec says "dest_path may be absolute or contain .. segments".
    # We must raise an error if it escapes the resolved destination root.
    # This usually means if dest_path is something like "/tmp/..", we should reject it.
    
    # Normalize the path
    normalized = os.path.normpath(dest_path)
    
    # If the normalized path is absolute, it's the root.
    if os.path.isabs(normalized):
        return normalized
    
    # If relative, we need a base. The spec doesn't explicitly give a base, but implies
    # we shouldn't escape the "resolved destination root".
    # In many secure implementations, if dest_path is relative, we resolve it relative to the cwd.
    # However, to be safe against .., we should check if the normalized path goes outside the cwd.
    # But the spec says "escapes the resolved destination root".
    # Let's assume the root is the directory where the script is run, or simply ensure the
    # normalized path doesn't have .. that resolves outside the intended scope.
    # The most strict interpretation: if dest_path is relative, resolve it. If the result
    # has any .. that would resolve outside the current directory, fail.
    # Actually, simpler: just ensure the final absolute path is what we expect.
    # If the user provides a relative path with .., we resolve it. If it goes outside the
    # current working directory, that's an escape.
    
    # Let's resolve the absolute path.
    abs_dest = os.path.abspath(normalized)
    
    # Check if it escapes the current directory (if dest_path was relative)
    # If dest_path was absolute, abs_dest should equal normalized (with .. resolved).
    # If abs_dest != normalized, it means .. was resolved to something else.
    # But wait, os.path.abspath on "/tmp/.." returns "/tmp". That's fine.
    # The danger is "/..".
    
    # Let's just ensure the absolute path is exactly the normalized path (with .. resolved)
    # and doesn't go above the "root".
    # Since we don't have a specific root other than the dest_path itself, we assume
    # the dest_path is the root.
    # We must ensure that after normalization, we don't have .. that resolves outside.
    # If dest_path is absolute, we just normalize.
    # If dest_path is relative, we resolve it. If the resolved path is outside cwd, fail.
    
    # To be safe, we can check if the normalized path starts with the cwd.
    cwd = os.getcwd()
    if not os.path.isabs(normalized):
        # If relative, ensure it doesn't escape cwd
        if not abs_dest.startswith(cwd):
            raise ValueError(f"Destination path escapes current directory: {abs_dest}")
    
    return abs_dest
