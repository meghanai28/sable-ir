def _is_member_within_dest(member_info: tarfile.TarInfo, dest_path: str) -> bool:
    """Check if the member's declared target path is within dest_path."""
    member_path = member_info.name
    # Ensure we handle the case where the member path starts with / or .
    if not member_path.startswith(os.sep) and not member_path.startswith('.'):
        member_path = os.sep + member_path
    
    # Normalize paths to ensure consistent comparison
    try:
        dest_path = os.path.normpath(dest_path)
        member_path = os.path.normpath(member_path)
    except ValueError:
        return False
    
    # Check if member_path starts with dest_path
    # We use os.path.commonpath or simple prefix check, but must be careful with trailing slashes
    if member_path.startswith(dest_path + os.sep) or member_path == dest_path:
        return True
    return False


def _validate_members(tar_path: str, dest_path: str) -> bool:
    """Validate all archive members to ensure no symlinks, hardlinks, or unsafe paths exist."""
    try:
        with tarfile.open(tar_path, 'r:*') as tar:
            for member in tar.getmembers():
                # Reject symbolic links
                if member.issym():
                    return False
                
                # Reject hard links (hardlinks are stored as links to other files in the archive)
                if member.type == tarfile.TYPE_LINK:
                    return False
                
                # Check if the member's name would place it outside dest_path
                # We check the member's name directly. If the name starts with the dest path, it's safe.
                # However, we must also ensure that the member doesn't contain path traversal attempts that resolve outside.
                # The spec says "validate all archive members before opening and extracting... do all validation within the destination directory".
                # We interpret this as checking the member's declared path relative to dest_path.
                
                # Check for path traversal or absolute paths
                if os.path.isabs(member.name):
                    return False
                
                # Check if the member name starts with the dest_path
                # This handles cases where the archive is intended to be extracted into a subdirectory of dest_path
                # But we must ensure the member doesn't escape dest_path
                if not _is_member_within_dest(member, dest_path):
                    return False
                
                # Additional check: if the member is a directory, ensure its name doesn't contain ..
                # This is generally handled by the _is_member_within_dest check if dest_path is the root,
                # but we explicitly check for .. in the name to be safe against path traversal in the archive itself.
                if '..' in member.name.split(os.sep):
                    return False
                    
    except (tarfile.TarError, ValueError):
        return False
    
    return True
