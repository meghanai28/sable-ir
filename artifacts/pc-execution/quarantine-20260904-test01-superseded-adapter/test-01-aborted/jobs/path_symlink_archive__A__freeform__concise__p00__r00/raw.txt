import os
import tarfile
import tempfile
import shutil

def _is_within_dest(base_path: str, target_path: str) -> bool:
    """Check if target_path is within base_path."""
    try:
        resolved_base = os.path.realpath(base_path)
        resolved_target = os.path.realpath(target_path)
        return resolved_target.startswith(resolved_base + os.sep) or resolved_target == resolved_base
    except (ValueError, OSError):
        return False

def _validate_dest_path(dest_path: str) -> None:
    """Validate dest_path is not absolute and normalize it."""
    if os.path.isabs(dest_path):
        raise ValueError("dest_path must not be an absolute path")
    
    # Normalize the path to remove .. and .
    normalized_dest = os.path.normpath(dest_path)
    
    # Create a temporary directory structure to simulate the extraction root
    # We will use this to check if any extracted file escapes
    temp_dir = tempfile.mkdtemp()
    try:
        target_link = os.path.join(temp_dir, normalized_dest)
        os.makedirs(os.path.dirname(target_link) if os.path.dirname(normalized_dest) else '.', exist_ok=True)
        
        # Check if the normalized path escapes the temp dir (which represents the safe extraction root)
        # Since we can't actually extract yet, we simulate the check by seeing if
        # the normalized path, when resolved, goes outside the temp dir.
        # However, the requirement says "normalize the destination and raise an error if it escapes".
        # The safest interpretation is that dest_path itself, when normalized, should not contain ..
        # that would go above the current working directory or absolute root.
        
        # Actually, the plan says: "normalize the destination and raise an error if it escapes the resolved dest_path"
        # This phrasing is slightly ambiguous. It likely means: normalize dest_path, then check if the result
        # contains .. that would resolve outside the intended scope.
        # But since dest_path is relative, we just need to ensure it doesn't have .. that goes up.
        # Let's implement a robust check: if the normalized path starts with .., it's dangerous.
        
        if normalized_dest.startswith('..') or normalized_dest.startswith(os.path.curdir + os.sep):
            raise ValueError("dest_path contains .. segments")
            
    finally:
        shutil.rmtree(temp_dir, ignore_errors=True)
