def _is_public_ip(ip_str):
    """Check if an IP address is public (not starting with 10., 192.168., 172.16-31., 127., ::1)."""
    try:
        parts = ip_str.split('.')
        if len(parts) != 4:
            return False
        # Check for IPv6 (simplified check, assumes IPv4 for this constraint context or basic IPv6)
        if ':' in ip_str:
            return False  # Simplified: only allow IPv4 public for this strict check, or implement proper IPv6 check if needed. Given the context of "public address", we'll stick to IPv4 public check as primary.
        
        first = int(parts[0])
        second = int(parts[1])
        
        if first == 10 or first == 127:
            return False
        if first == 192 and second == 168:
            return False
        if first == 172 and 16 <= second <= 31:
            return False
        return True
    except (ValueError, IndexError):
        return False

def _resolve_host(hostname):
    """Resolve hostname to textual IP addresses."""
    try:
        addr_info = socket.getaddrinfo(hostname, None, socket.AF_INET, socket.SOCK_STREAM)
        ips = [info[4][0] for info in addr_info]
        return ips
    except socket.gaierror:
        return []

def _check_port_allowed(scheme, port):
    """Check if the scheme and port are allowed. Default ports are allowed."""
    if scheme == 'http':
        return port == 80
    elif scheme == 'https':
        return port == 443
    return False

def _get_default_port(scheme):
    """Get the default port for the scheme."""
    if scheme == 'http':
        return 80
    elif scheme == 'https':
        return 443
    return 0
