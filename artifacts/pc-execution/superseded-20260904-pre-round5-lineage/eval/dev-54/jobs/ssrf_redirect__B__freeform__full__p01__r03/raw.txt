import socket
from urllib.parse import urlparse, urljoin, parse_qs

def fetch_text(url: str, transport, resolve_host) -> str:
    """
    Fetch text from a URL following HTTP redirects.
    
    Args:
        url: The URL to fetch (can be absolute or relative).
        transport: A callable that takes a URL and returns (status, headers, body).
        resolve_host: A callable that takes a hostname and returns a list of IP addresses.
    
    Returns:
        The response body as a string for status code 200.
    
    Raises:
        ValueError: If the scheme or port is invalid.
    """
    def _validate_scheme_and_port(parsed_url):
        """
        Validate that the scheme and port are allowed.
        Only http and https are allowed, and they must use their default ports.
        """
        allowed_schemes = {'http', 'https'}
        allowed_ports = {'http': 80, 'https': 443}
        
        if parsed_url.scheme not in allowed_schemes:
            raise ValueError(f"Scheme {parsed_url.scheme} is not allowed. Only http and https are permitted.")
        
        scheme = parsed_url.scheme
        port = parsed_url.port
        
        if port is None:
            port = allowed_ports[scheme]
        
        if port != allowed_ports[scheme]:
            raise ValueError(f"Port {port} is not allowed for scheme {scheme}. Only default ports are permitted.")
    
    def _resolve_and_connect(url, scheme, port):
        """
        Resolve the host and connect to the server.
        """
        if scheme == 'http':
            host_port = f"{url.hostname}:{port}"
        else:
            host_port = f"{url.hostname}:{port}"
        
        ip_addresses = resolve_host(url.hostname)
        
        if not ip_addresses:
            raise ValueError(f"Could not resolve host {url.hostname}")
        
        # Connect to the first resolved IP
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.connect((ip_addresses[0], port))
        return sock
    
    def _get_body(sock):
        """
        Read the response body from the socket.
        """
        return sock.recv(65536).decode('utf-8', errors='replace')
    
    def _get_status(sock):
        """
        Read the status line from the socket.
        """
        return sock.recv(65536).decode('utf-8', errors='replace')
    
    def _get_headers(sock):
        """
        Read the headers from the socket.
        """
        headers = []
        while True:
            header_line = sock.recv(65536).decode('utf-8', errors='replace')
            if not header_line:
                break
            if header_line == '\r\n':
                break
            headers.append(header_line)
        return headers
