import os

def read_report(filename: str, reports_root: str) -> str:
    """
    Read and return the requested UTF-8 report beneath reports_root.

    when the request is disallowed (e.g., path traversal, absolute paths, or symlinks).
    """
    # Normalize the root path to an absolute path to prevent directory traversal via relative paths
    resolved_root = os.path.normpath(os.path.abspath(reports_root))
    
    # Check if filename is absolute

        raise ValueError("Absolute filenames are not allowed")
