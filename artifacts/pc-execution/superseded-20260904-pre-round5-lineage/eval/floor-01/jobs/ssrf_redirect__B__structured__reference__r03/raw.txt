def _is_global_public_address(addr: str) -> bool:
    """Check if an IP address is a global public address (not localhost, link-local, etc.)."""
    try:
        ip = socket.inet_aton(addr)
        # Check for private/reserved ranges
        # 127.0.0.0/8 - Loopback
        # 10.0.0.0/8 - Private
        # 172.16.0.0/12 - Private
        # 192.168.0.0/16 - Private
        # 169.254.0.0/16 - Link-local
        # 0.0.0.0/32 - Default route
        # 224.0.0.0/4 - Multicast
        # 240.0.0.0/4 - Reserved
        # ::1, fe80::, etc. for IPv6 if needed, but transport likely returns IPv4 here or we check both.
        # For simplicity in this context, we assume standard IPv4 checks or a simple heuristic.
        # A robust check would parse the address into a tuple and check octets.
        parts = addr.split('.')
        if len(parts) != 4:
            return False
        try:
            octets = [int(p) for p in parts]
        except ValueError:
            return False
        
        # Loopback
        if octets[0] == 127:
            return False
        # Link-local
        if octets[0] == 169 and octets[1] == 254:
            return False
        # Multicast/Reserved
        if octets[0] >= 224:
            return False
        # Private ranges
        if octets[0] == 10:
            return False
        if octets[0] == 172 and 16 <= octets[1] <= 31:
            return False
        if octets[0] == 192 and octets[1] == 168:
            return False
        
        return True
    except (socket.error, OSError):
        return False

def _validate_destination(url: str, resolve_host: callable) -> Tuple[str, int]:
    """Validate the URL destination according to the plan.
    Returns (scheme, port) or raises ValueError.
    """
    parsed = urllib.parse.urlparse(url)
    scheme = parsed.scheme.lower()
    port = parsed.port
    
    # Check for username/password
    if parsed.username or parsed.password:
        raise ValueError("URL contains username or password")
    
    # Check scheme and port
    allowed_schemes = ['http', 'https']
    if scheme not in allowed_schemes:
        raise ValueError(f"Disallowed scheme: {scheme}")
    
    # Check port
    if scheme == 'http':
        if port is not None and port != 80:
            raise ValueError(f"HTTP must use port 80 or be omitted, got {port}")
    elif scheme == 'https':
        if port is not None and port != 443:
            raise ValueError(f"HTTPS must use port 443 or be omitted, got {port}")
    
    # Resolve hostname
    hostname = parsed.hostname
    if not hostname:
        raise ValueError("Invalid hostname in URL")
    
    try:
        addresses = resolve_host(hostname)
    except Exception as e:
        raise ValueError(f"Failed to resolve hostname {hostname}: {e}")
    
    if not addresses:
        raise ValueError(f"Resolve failed for {hostname}")
    
    # Check if any address is global public
    # The plan says "raise ValueError if it yields no addresses or any address is not a global public address"
    # This implies ALL addresses must be public? Or at least one?
    # "any address is not a global public address" -> if ANY is bad, raise.
    # So we must ensure all returned addresses are public.
    for addr in addresses:
        if not _is_global_public_address(addr):
            raise ValueError(f"Non-public address found for {hostname}: {addr}")
    
    return scheme, port
