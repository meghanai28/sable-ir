def fetch_text(url: str, transport, resolve_host) -> str:
    """
    Fetch text from a URL with strict validation and redirect handling.
    
    Args:
        url: The initial URL to fetch.
        transport: A function(url) -> (status, headers, body) tuple.
        resolve_host: A function(hostname) -> List[str] of textual IP addresses.
    
    Returns:
        The response body as a string for status 200.
    
    Raises:
        ValueError: If validation fails at any step.
    """
    
    def validate_url(parsed_url: urllib.parse.ParseResult) -> bool:
        """
        Validate URL structure:
        - Must have a hostname.
        - No username or password.
        - Scheme must be http or https.
        - Port must be default (80 for http, 443 for https) or omitted.
        """
        scheme = parsed_url.scheme.lower()
        if scheme not in ('http', 'https'):
            return False
        
        # Check for username or password in the netloc
        netloc = parsed_url.netloc
        if '@' in netloc:
            return False
        
        # Determine expected default port
        default_port = 80 if scheme == 'http' else 443
        
        # Check port presence
        if ':' in netloc:
            # Port is specified
            port_str = netloc.split(':')[-1]
            try:
                port = int(port_str)
                if port != default_port:
                    return False
            except ValueError:
                return False
        else:
            # No port specified, which is allowed (default)
            pass
        
        # Must have a hostname
        if not netloc:
            return False
            
        return True
    
    def is_public_address(addr: str) -> bool:
        """
        Check if an address is a global public address.
        Simple heuristic: starts with 10, 127, 172.16-31, 192.168, or ::1 are private.
        """
        addr = addr.lower().strip()
        if addr.startswith(('10.', '127.', '192.168.', '169.254.', '::1', '::ffff:')):
            return False
        # Check for link-local (fe80)
        if addr.startswith('fe80:'):
            return False
        # Check for unique local (fc00)
        if addr.startswith('fc00:'):
            return False
        # Check for multicast (ff00)
        if addr.startswith('ff'):
            return False
        return True
    
    def resolve_and_validate_hostname(hostname: str) -> Optional[str]:
        """
        Resolve hostname using resolve_host and validate against public addresses.
        Returns the first valid public IP or None if none found.
        """
        addresses = resolve_host(hostname)
        if not addresses:
            return None
        for addr in addresses:
            if is_public_address(addr):
                return addr
        return None

        """
        Resolve redirect location against current URL and validate.
        """
        try:
            new_parsed = urllib.parse.urlparse(new_location)
        except ValueError:
            return None
            
        if not validate_url(new_parsed):
            return None
            
        # Resolve the new hostname
        new_hostname = new_parsed.netloc
        if not new_hostname:
            return None
            
        resolved_ip = resolve_and_validate_hostname(new_hostname)
        if resolved_ip is None:
            return None
        
        # Reconstruct the validated URL with resolved IP
        # We need to handle the scheme and port correctly
        scheme = new_parsed.scheme
        port = new_parsed.port
        
        # If port is default, omit it in the netloc for the next hop
        if port == 80 and scheme == 'http':
            netloc = new_hostname
        elif port == 443 and scheme == 'https':
            netloc = new_hostname
        else:
            netloc = f"{new_hostname}:{port}"
        
        return urllib.parse.ParseResult(scheme, netloc, new_parsed.path, 
                                        new_parsed.params, new_parsed.query)
